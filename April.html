import java.util.Scanner;
import java.util.Arrays;

public class DownloadTimeTracker {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        double[] times = new double[12];
        double total = 0;
        double fastest, slowest;
        int count = 0;

        System.out.println(" DOWNLOAD TIME TRACKER ");

        // Input 12 download times
        for (int i = 0; i < times.length; i++) {
            do {
                System.out.print("Enter download time " + (i + 1) + " (seconds): ");
                times[i] = input.nextDouble();

                if (times[i] <= 0) {
                    System.out.println("Please enter a positive value.");
                }
            } while (times[i] <= 0);

            total += times[i];

            if (times[i] < 30) {
                count++;
            }
        }

        // Calculate average
        double average = total / times.length;

        // Find fastest and slowest
        fastest = times[0];
        slowest = times[0];

        for (int i = 1; i < times.length; i++) {
            if (times[i] < fastest) {
                fastest = times[i];
            }

            if (times[i] > slowest) {
                slowest = times[i];
            }
        }

        // Sort download times
        Arrays.sort(times);

        // Display results
        System.out.println("\n DOWNLOAD RESULTS ");
        System.out.printf("Average Time: %.2f seconds%n", average);
        System.out.println("Fastest Download: " + fastest + " seconds");
        System.out.println("Slowest Download: " + slowest + " seconds");
        System.out.println("Downloads below 30 seconds: " + count);

        System.out.println("\nDownload Times in Ascending Order:");
        for (double time : times) {
            System.out.println(time + " seconds");
        }

        input.close();
    }
}
